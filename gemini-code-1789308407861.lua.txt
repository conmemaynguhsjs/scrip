--// UTILITY FUNCTIONS (COMPREHENSIVE UTILS)
local Utility = {}

function Utility.Clamp(val, min, max)
    return math.max(min, math.min(max, val))
end

function Utility.GetHumanoid(player)
    local char = player and player.Character
    if char then
        return char:FindFirstChildOfClass("Humanoid")
    end
end

function Utility.GetRootPart(player)
    local char = player and player.Character
    if char then
        return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
    end
end

function Utility.GetDistanceFromLocal(targetPlayer)
    local myRoot = Utility.GetRootPart(LocalPlayer)
    local targetRoot = Utility.GetRootPart(targetPlayer)
    if myRoot and targetRoot then
        return (myRoot.Position - targetRoot.Position).Magnitude
    end
    return math.huge
end

function Utility.IsWhitelisted(name)
    return STATE.WhitelistPlayers[name] == true
end

function Utility.SetWhitelisted(name, state)
    STATE.WhitelistPlayers[name] = state
end

local function pushLog(level, msg)
    table.insert(STATE.Logs, {
        Time = os.date("%H:%M:%S"),
        Level = level,
        Message = msg
    })
    if #STATE.Logs > 300 then
        table.remove(STATE.Logs, 1)
    end
end

local function setStatus(status, msg)
    STATE.Status = status
    STATE.LastMessage = msg
    pushLog(status, msg)
end

local function make(className, properties, parent)
    local obj = Instance.new(className)
    for k, v in pairs(properties) do
        obj[k] = v
    end
    if parent then
        obj.Parent = parent
    end
    return obj
end

local function corner(parent, radius)
    return make("UICorner", { CornerRadius = UDim.new(0, radius) }, parent)
end

local function stroke(parent, color, thickness)
    return make("UIStroke", { Color = color, Thickness = thickness or 1 }, parent)
end

local function tween(obj, props, duration)
    local info = TweenInfo.new(duration or 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    local t = TweenService:Create(obj, info, props)
    t:Play()
    return t
end

local function connect(name, signal, callback)
    if CONNECTIONS[name] then
        CONNECTIONS[name]:Disconnect()
    end
    CONNECTIONS[name] = signal:Connect(callback)
end

function Movement:SetTPWalk(enabled)
    STATE.TPWalkEnabled = enabled
    if tpWalkConn then tpWalkConn:Disconnect() tpWalkConn = nil end
    if enabled then
        tpWalkConn = RunService.RenderStepped:Connect(function(dt)
            local char = LocalPlayer.Character
            local root = Utility.GetRootPart(LocalPlayer)
            local hum = Utility.GetHumanoid(LocalPlayer)
            if char and root and hum and hum.MoveDirection.Magnitude > 0 then
                root.CFrame = root.CFrame + (hum.MoveDirection * (STATE.WalkValue * dt))
            end
        end)
        setStatus("RUNNING", "TPWalk Hack Enabled")
    else
        setStatus("READY", "TPWalk Disabled")

end

        bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)

            local delta = input.Position - dragStart
            UI.Main.Position = UDim2.new(

                startPosition.Y.Scale, startPosition.Y.Offset + delta.Y

        end
    end)

--// RUNTIME & INITIALIZATION BINDINGS
local function bindRuntime()
    connect("Render", RunService.RenderStepped, function()
        if STATE.Destroyed then return end
        ESP:UpdatePlayers()
        updateStatus()
    end)

    connect("Input", UserInputService.InputBegan, function(input, processed)
        if processed or STATE.Destroyed then return end
        if input.KeyCode == Enum.KeyCode.RightShift and UI.Gui then
            UI.Gui.Enabled = not UI.Gui.Enabled
        end
    end)
end

local function cleanup()
    if STATE.Destroyed then return end
    STATE.Destroyed = true
    STATE.AutoStealRunning = false
    STATE.AutoSellRunning = false
    STATE.AutoRebirthRunning = false
    STATE.BatAuraRunning = false

    for key, connection in pairs(CONNECTIONS) do
        if connection then connection:Disconnect() end
        CONNECTIONS[key] = nil
    end

    ESP:Clear()
    TestAPI:Disconnect()

    if UI.Gui then
        UI.Gui:Destroy()
        UI.Gui = nil
    end
end

local function initialize()
    if STATE.Initialized then return end
    STATE.Initialized = true
    STATE.Destroyed = false

    createUI()
    bindRuntime()

    pushLog("INFO", "Steal an Egg Expanded Exploit Suite v" .. CONFIG.Version .. " initialized successfully.")
    setStatus("READY", "Full exploit hub operational")
end

    AutoRebirth = AutoRebirth,
    Combat = Combat,
    ESP = ESP,
    Security = Security,
    Init = initialize,
    Cleanup = cleanup,
}
