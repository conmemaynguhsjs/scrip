local function AutoFindRemote(keywordList)
    for _, desc in ipairs(game:GetDescendants()) do
        if desc:IsA("RemoteEvent") or desc:IsA("RemoteFunction") then
            local name = desc.Name:lower()
            for _, kw in ipairs(keywordList) do
                if name:find(kw:lower()) then
                    return desc
                end
            end
        end
    end
    return nil
end

    local function LoadPetIntoPreview(petModel)
        CleanupPreview()

            if petModel and petModel:IsA("Model") then
                local clonedModel = petModel:Clone()

                local cf, size = clonedModel:GetBoundingBox()
                local maxDim = math.max(size.X, size.Y, size.Z)
                if maxDim <= 0 then maxDim = 5 end
                local camDist = maxDim * 2.2
                
                Camera.CFrame = CFrame.new(cf.Position + Vector3.new(0, maxDim * 0.3, camDist), cf.Position)
                
                local currentAngle = 0
                PreviewRotationConnection = RunService.RenderStepped:Connect(function(dt)
                    pcall(function()
                        if clonedModel and clonedModel.Parent then
                            currentAngle = currentAngle + (dt * 45)
                            local pivot = clonedModel:GetPivot()
                            local currentPos = pivot.Position
                            clonedModel:PivotTo(CFrame.new(currentPos) * CFrame.Angles(0, math.rad(currentAngle), 0))
                        else
                            CleanupPreview()
                        end
                    end)
                end)
            end
        end)
    end

    local function GetPetsForEgg(eggInstance)
        local petsList = {}
        pcall(function()
            if not eggInstance or not eggInstance:IsA("Model") then return end

            local function validateAndAdd(model)
                if not model or not model:IsA("Model") then return end
                local n = model.Name:lower()
                if n:find("vfx") or n:find("effect") or n:find("decoration") or n:find("display") or n:find("dummy") or n:find("shell") or n:find("particle") then
                    return
                end
                if model:FindFirstChildWhichIsA("BasePart", true) then
                    if not table.find(petsList, model) then
                        table.insert(petsList, model)
                    end
                end
            end

            for _, child in pairs(eggInstance:GetChildren()) do
                if child:IsA("Model") and child ~= eggInstance then
                    validateAndAdd(child)
                elseif child:IsA("Folder") then
                    for _, subChild in pairs(child:GetChildren()) do
                        if subChild:IsA("Model") then
                            validateAndAdd(subChild)
                        end
                    end
                end
            end
        end)
        return petsList
    end

    local function MatchEggFilter(eggObj)
        local success, result = pcall(function()
            local nameLower = eggObj.Name:lower()

            if FilterConfig.SelectedMap ~= "Tất cả Map" then
                local mapKey = FilterConfig.SelectedMap:lower():gsub("map ", "")
                if not nameLower:find(mapKey) and not (eggObj.Parent and eggObj.Parent.Name:lower():find(mapKey)) then
                    return false
                end
            end

            if FilterConfig.FilterMutation then
                if not nameLower:find("mut") and not nameLower:find("mutation") and not eggObj:FindFirstChild("Mutation") then
                    return false
                end
            end

            if FilterConfig.FilterHuge then
                if not nameLower:find("huge") and not eggObj:FindFirstChild("Huge") then
                    return false
                end
            end

            if FilterConfig.SelectedCategory ~= "Tất cả" then
                local catKey = FilterConfig.SelectedCategory:lower()
                if catKey:find("thường") and (nameLower:find("special") or nameLower:find("event")) then return false end
                if catKey:find("đặc biệt") and not nameLower:find("special") then return false end
                if catKey:find("sự kiện") and not nameLower:find("event") then return false end
                if catKey:find("boss") and not nameLower:find("boss") then return false end
            end

            return true
        end)
        if success then return result else return true end
    end

SelectedEggInstance = nil
SelectedPetModel = nil
local CollectAttempted = false
local ReturnSpawnPosition = nil

local function GetEggTargetPosition(egg)
    if not egg or not egg.Parent then return nil end
    local success, pos = pcall(function()
        if egg:IsA("Model") then
            if egg.PrimaryPart then
                return egg.PrimaryPart.Position
            end
            local okPivot, pivotCF = pcall(function() return egg:GetPivot() end)
            if okPivot and pivotCF then
                return pivotCF.Position
            end
            local okBox, boxCF, _ = pcall(function() return egg:GetBoundingBox() end)
            if okBox and boxCF then
                return boxCF.Position
            end
            for _, desc in pairs(egg:GetDescendants()) do
                if desc:IsA("BasePart") then
                    local n = desc.Name:lower()
                    if not n:find("vfx") and not n:find("particle") and not n:find("effect") and not n:find("decoration") then
                        return desc.Position
                    end
                end
            end
            local anyPart = egg:FindFirstChildWhichIsA("BasePart", true)
            if anyPart then return anyPart.Position end
        elseif egg:IsA("BasePart") then
            return egg.Position
        end
        return nil
    end)
    if success then return pos else return nil end
end

local function GetPlayerSpawnPosition()
    local success, pos = pcall(function()
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("SpawnLocation") then
                if obj.TeamColor == LocalPlayer.TeamColor then
                    return obj.Position + Vector3.new(0, 3, 0)
                end
            end
        end
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("SpawnLocation") then
                if obj.Neutral then
                    return obj.Position + Vector3.new(0, 3, 0)
                end
            end
        end
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("SpawnLocation") then
                return obj.Position + Vector3.new(0, 3, 0)
            end
        end
        return nil
    end)
    if success then return pos else return nil end
end

local function StopFly()
    pcall(function()
        if FlyConnection then
            FlyConnection:Disconnect()
            FlyConnection = nil
        end
        if FlyVelocity then
            FlyVelocity:Destroy()
            FlyVelocity = nil
        end
        if FlyGyro then
            FlyGyro:Destroy()
            FlyGyro = nil
        end
        if Humanoid then
            Humanoid.PlatformStand = false
        end
    end)
end

local function StartFly()
    if FlyConnection then return end
    
    local flyStartTime = tick()
    local currentTaskState = "FreeFly"

    if State.FlyToSelectedEgg then
        currentTaskState = "FlyToEgg"
        CollectAttempted = false
    elseif State.ReturnToSpawn then
        currentTaskState = "ReturnToSpawn"
    elseif State.TrueFly then
        currentTaskState = "FreeFly"
    end

    FlyConnection = RunService.RenderStepped:Connect(function()
        pcall(function()
            if not State.TrueFly and not State.FlyToSelectedEgg and not State.ReturnToSpawn then
                StopFly()
                State.FlyToSelectedEgg = false
                State.ReturnToSpawn = false
                State.TrueFly = false
                CollectAttempted = false
                ReturnSpawnPosition = nil
                return
            end

            if not Character or not Humanoid or not RootPart or Humanoid.Health <= 0 then
                return
            end

            Humanoid.PlatformStand = true

            if not FlyVelocity or not FlyVelocity.Parent then
                FlyVelocity = Instance.new("BodyVelocity")
                FlyVelocity.Name = "FlyVelocity"
                FlyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                FlyVelocity.Velocity = Vector3.new(0, 0, 0)
                FlyVelocity.Parent = RootPart
            end

            if not FlyGyro or not FlyGyro.Parent then
                FlyGyro = Instance.new("BodyGyro")
                FlyGyro.Name = "FlyGyro"
                FlyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                FlyGyro.CFrame = RootPart.CFrame
                FlyGyro.Parent = RootPart
            end

            local camera = workspace.CurrentCamera

            if currentTaskState == "FlyToEgg" or currentTaskState == "ReturnToSpawn" then
                if tick() - flyStartTime > 15 then
                    StopFly()
                    State.FlyToSelectedEgg = false
                    State.ReturnToSpawn = false
                    State.TrueFly = false
                    CollectAttempted = false
                    ReturnSpawnPosition = nil
                    return
                end
            end

            if currentTaskState == "FlyToEgg" then
                if not SelectedEggInstance or not SelectedEggInstance.Parent then
                    StopFly()
                    State.FlyToSelectedEgg = false
                    State.ReturnToSpawn = false
                    State.TrueFly = false
                    CollectAttempted = false
                    ReturnSpawnPosition = nil
                    return
                end

                local targetPos = GetEggTargetPosition(SelectedEggInstance)
                if not targetPos then
                    StopFly()
                    State.FlyToSelectedEgg = false
                    State.ReturnToSpawn = false
                    State.TrueFly = false
                    CollectAttempted = false
                    ReturnSpawnPosition = nil
                    return
                end

                local dir = (targetPos - RootPart.Position)
                if dir.Magnitude > 6 then
                    FlyVelocity.Velocity = dir.Unit * State.FlySpeed
                    FlyGyro.CFrame = camera.CFrame
                else
                    FlyVelocity.Velocity = Vector3.zero

                    if not CollectAttempted then
                        CollectAttempted = true

                        local successCall, apiSuccess, apiMsg = pcall(function()
                            return GameAPI.CollectSelectedEgg(SelectedEggInstance)
                        end)

                        if not successCall or apiSuccess == false then
                            StopFly()
                            State.FlyToSelectedEgg = false
                            State.ReturnToSpawn = false
                            State.TrueFly = false
                            CollectAttempted = false
                            ReturnSpawnPosition = nil
                            return
                        else
                            ReturnSpawnPosition = GetPlayerSpawnPosition()
                            if not ReturnSpawnPosition then
                                StopFly()
                                State.FlyToSelectedEgg = false
                                State.ReturnToSpawn = false
                                State.TrueFly = false
                                CollectAttempted = false
                                ReturnSpawnPosition = nil
                                return
                            end

                            State.FlyToSelectedEgg = false
                            State.ReturnToSpawn = true
                            currentTaskState = "ReturnToSpawn"
                            flyStartTime = tick()
                        end
                    end
                end

            elseif currentTaskState == "ReturnToSpawn" then
                if not ReturnSpawnPosition then
                    ReturnSpawnPosition = GetPlayerSpawnPosition()
                end

                if not ReturnSpawnPosition then
                    StopFly()
                    State.FlyToSelectedEgg = false
                    State.ReturnToSpawn = false
                    State.TrueFly = false
                    CollectAttempted = false
                    ReturnSpawnPosition = nil
                    return
                end

                local dir = (ReturnSpawnPosition - RootPart.Position)
                if dir.Magnitude > 5 then
                    FlyVelocity.Velocity = dir.Unit * State.FlySpeed
                    FlyGyro.CFrame = camera.CFrame
                else
                    StopFly()
                    State.FlyToSelectedEgg = false
                    State.ReturnToSpawn = false
                    State.TrueFly = false
                    CollectAttempted = false
                    ReturnSpawnPosition = nil
                end

            elseif currentTaskState == "FreeFly" then
                local moveDir = Humanoid.MoveDirection
                local velocity = Vector3.zero

                if moveDir.Magnitude > 0 then
                    velocity = moveDir * State.FlySpeed
                end

                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                    velocity = velocity + Vector3.new(0, State.FlySpeed, 0)
                end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
                    velocity = velocity - Vector3.new(0, State.FlySpeed, 0)
                end

                FlyVelocity.Velocity = velocity
                FlyGyro.CFrame = camera.CFrame
            end
        end)
    end)
end
