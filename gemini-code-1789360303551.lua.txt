local function AutoFindRemote(keywordList)
    for _, desc in ipairs(game:GetDescendants()) do
        if desc:IsA("RemoteEvent") or desc:IsA("RemoteFunction") then
            local name = desc.Name:lower()
            for _, kw in ipairs(keywordList) do
                if name:find(kw:lower()) then
                    return desc
                end
            end
        end
    end
    return nil
end

GameAPI.Rebirth = AutoFindRemote({"rebirth", "prestige", "evolution", "upgrade"})
GameAPI.CollectDrop = AutoFindRemote({"collectdrop", "pickup", "getcoin", "loot", "money", "gem"})
GameAPI.FarmBoss = AutoFindRemote({"boss", "attackboss", "fightboss", "rift", "monster"})

GameAPI.Treadmill = AutoFindRemote({"treadmill", "gym", "train", "workout"})

GameAPI.CollectSelectedEgg = function(egg)
    local remote = AutoFindRemote({"collectegg", "stealand", "takeegg", "eggcollect", "steal"})
    if remote then
        if remote:IsA("RemoteEvent") then
            remote:FireServer(egg)
        elseif remote:IsA("RemoteFunction") then
            remote:InvokeServer(egg)
        end
        return true, "Đã gửi yêu cầu thành công"
    end
    return false, "Không tìm thấy Remote phù hợp"
end

GameAPI.CollectAllMutatedHuge = function()
    local remote = AutoFindRemote({"mutated", "huge", "collectall", "claimall", "claim"})
    if remote then
        if remote:IsA("RemoteEvent") then
            remote:FireServer()
        end
        return true
    end
    return false
end

local CollectAttempted = false
local ReturnSpawnPosition = nil

local function GetEggTargetPosition(egg)
    if not egg or not egg.Parent then return nil end
    local success, pos = pcall(function()
        if egg:IsA("Model") then
            if egg.PrimaryPart then return egg.PrimaryPart.Position end
            local pivotCF = egg:GetPivot()
            if pivotCF then return pivotCF.Position end
            local part = egg:FindFirstChildWhichIsA("BasePart", true)
            if part then return part.Position end
        elseif egg:IsA("BasePart") then
            return egg.Position
        end
        return nil
    end)
    return success and pos or nil
end

local function GetPlayerSpawnPosition()
    local success, pos = pcall(function()
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("SpawnLocation") and (obj.TeamColor == LocalPlayer.TeamColor or obj.Neutral) then
                return obj.Position + Vector3.new(0, 3, 0)
            end
        end
        return nil
    end)
    return success and pos or nil
end

local function StopFly()
    pcall(function()
        if FlyConnection then FlyConnection:Disconnect() FlyConnection = nil end
        if FlyLinearVelocity then FlyLinearVelocity:Destroy() FlyLinearVelocity = nil end
        if FlyAlignOrientation then FlyAlignOrientation:Destroy() FlyAlignOrientation = nil end
        if FlyAttachment then FlyAttachment:Destroy() FlyAttachment = nil end
        if Humanoid then Humanoid.PlatformStand = false end
    end)
end

local function StartFly()
    if FlyConnection then return end
    local flyStartTime = tick()
    local currentTaskState = State.FlyToSelectedEgg and "FlyToEgg" or (State.ReturnToSpawn and "ReturnToSpawn" or "FreeFly")
    if State.FlyToSelectedEgg then CollectAttempted = false end

    pcall(function()
        if not RootPart then return end
        FlyAttachment = Instance.new("Attachment", RootPart)
        FlyLinearVelocity = Instance.new("LinearVelocity", RootPart)
        FlyLinearVelocity.Attachment0 = FlyAttachment
        FlyLinearVelocity.MaxForce = math.huge
        FlyLinearVelocity.RelativeTo = Enum.ActuatorRelativeTo.World
        FlyLinearVelocity.VectorVelocity = Vector3.zero

        FlyAlignOrientation = Instance.new("AlignOrientation", RootPart)
        FlyAlignOrientation.Attachment0 = FlyAttachment
        FlyAlignOrientation.Mode = Enum.OrientationAlignMode.OneAttachment
        FlyAlignOrientation.MaxTorque = math.huge
        FlyAlignOrientation.Responsiveness = 300
    end)

    FlyConnection = RunService.RenderStepped:Connect(function()
        pcall(function()
            if not State.TrueFly and not State.FlyToSelectedEgg and not State.ReturnToSpawn then
                StopFly()
                return
            end
            if not Character or not Humanoid or not RootPart or Humanoid.Health <= 0 then return end
            Humanoid.PlatformStand = true
            local camera = workspace.CurrentCamera
            FlyAlignOrientation.CFrame = camera.CFrame

            if currentTaskState == "FlyToEgg" then
                if not SelectedEggInstance or not SelectedEggInstance.Parent then StopFly() State.FlyToSelectedEgg = false return end
                local targetPos = GetEggTargetPosition(SelectedEggInstance)
                if not targetPos then StopFly() State.FlyToSelectedEgg = false return end

                local dir = (targetPos - RootPart.Position)
                if dir.Magnitude > 6 then
                    FlyLinearVelocity.VectorVelocity = dir.Unit * State.FlySpeed
                else
                    FlyLinearVelocity.VectorVelocity = Vector3.zero
                    if not CollectAttempted then
                        CollectAttempted = true
                        GameAPI.CollectSelectedEgg(SelectedEggInstance)
                        ReturnSpawnPosition = GetPlayerSpawnPosition()
                        State.FlyToSelectedEgg = false
                        State.ReturnToSpawn = true
                        currentTaskState = "ReturnToSpawn"
                        flyStartTime = tick()
                    end
                end
            elseif currentTaskState == "ReturnToSpawn" then
                if not ReturnSpawnPosition then ReturnSpawnPosition = GetPlayerSpawnPosition() end
                if not ReturnSpawnPosition then StopFly() return end
                local dir = (ReturnSpawnPosition - RootPart.Position)
                if dir.Magnitude > 5 then
                    FlyLinearVelocity.VectorVelocity = dir.Unit * State.FlySpeed
                else
                    StopFly()
                    State.FlyToSelectedEgg = false
                    State.ReturnToSpawn = false
                    State.TrueFly = false
                    CollectAttempted = false
                end
            elseif currentTaskState == "FreeFly" then
                local moveDir = Humanoid.MoveDirection
                local velocity = Vector3.zero
                if moveDir.Magnitude > 0 then
                    local camCF = camera.CFrame
                    velocity = (camCF:VectorToWorldSpace(Vector3.new(moveDir.X, 0, moveDir.Z))).Unit * State.FlySpeed
                end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then velocity = velocity + Vector3.new(0, State.FlySpeed, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then velocity = velocity - Vector3.new(0, State.FlySpeed, 0) end
                FlyLinearVelocity.VectorVelocity = velocity
            end
        end)
    end)
end

                local shortestDist = math.huge
                for _, obj in pairs(workspace:GetDescendants()) do
                    pcall(function()
                        if obj:IsA("Model") and (obj.Name:lower().find("boss") or obj.Name:lower().find("rift") or obj.Name:lower().find("monster")) then
                            local hum = obj:FindFirstChildOfClass("Humanoid")
                            local primary = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                            if hum and hum.Health > 0 and primary then
                                local dist = (primary.Position - RootPart.Position).Magnitude
                                if dist < shortestDist then shortestDist = dist targetBoss = obj end
                            end
                        end
                    end)
                end
                if targetBoss and GameAPI.FarmBoss then

                    if primary then
                        RootPart.CFrame = primary.CFrame + Vector3.new(0, 6, 4)

                    end
                end
            end
        end)
        task.wait(0.3)
    end

    while true do
        pcall(function()
            if State.ESPPlayer then
                for _, plr in pairs(Players:GetPlayers()) do
                    if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("Head") then
                        local head = plr.Character.Head
                        if not activePlayerESPs[plr.Name] then
                            local bg = Instance.new("BillboardGui", ESPFolder)
                            bg.Size, bg.AlwaysOnTop, bg.Adornee = UDim2.new(0, 120, 0, 40), true, head
                            bg.StudsOffset = Vector3.new(0, 3, 0)
                            local lbl = Instance.new("TextLabel", bg)
                            lbl.Size, lbl.BackgroundTransparency, lbl.TextColor3 = UDim2.new(1, 0, 1, 0), true, Color3.fromRGB(0, 255, 255)
                            lbl.Font, lbl.TextSize = Enum.Font.GothamBold, 10
                            activePlayerESPs[plr.Name] = {Gui = bg, Label = lbl}
                        else
                            local dist = RootPart and math.floor((head.Position - RootPart.Position).Magnitude) or 0
                            activePlayerESPs[plr.Name].Label.Text = string.format("%s [%dm]", plr.Name, dist)
                        end
                    elseif activePlayerESPs[plr.Name] then
                        activePlayerESPs[plr.Name].Gui:Destroy()
                        activePlayerESPs[plr.Name] = nil
                    end
                end
            else
                for _, d in pairs(activePlayerESPs) do if d.Gui then d.Gui:Destroy() end end
                activePlayerESPs = {}
            end
        end)
        task.wait(0.5)
    end
end)
